package org.example.aispringboot.service;

import cn.hutool.core.util.StrUtil;
import com.baomidou.mybatisplus.core.conditions.query.LambdaQueryWrapper;
import com.baomidou.mybatisplus.extension.plugins.pagination.Page;
import org.example.aispringboot.dto.command.EmotionDiaryQueryDTO;
import org.example.aispringboot.dto.response.EmotionDiaryResponseDTO;
import org.example.aispringboot.entity.EmotionDiary;
import org.example.aispringboot.entity.User;
import org.example.aispringboot.mapper.EmotionDiaryMapper;
import org.example.aispringboot.mapper.UserMapper;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;

import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.stream.Collectors;

@Service
public class EmotionDiaryService {

    @Autowired
    private EmotionDiaryMapper emotionDiaryMapper;

    @Autowired
    private UserMapper userMapper;

    public Map<String, Object> queryDiaries(EmotionDiaryQueryDTO queryDTO) {
        int currentPage = Integer.parseInt(queryDTO.getCurrent());
        int size = Integer.parseInt(queryDTO.getSize());

        LambdaQueryWrapper<EmotionDiary> wrapper = new LambdaQueryWrapper<>();

        if (StrUtil.isNotBlank(queryDTO.getUserId())) {
            wrapper.eq(EmotionDiary::getUserId, Long.parseLong(queryDTO.getUserId()));
        }
        if (StrUtil.isNotBlank(queryDTO.getMinMoodScore())) {
            wrapper.ge(EmotionDiary::getMoodScore, Integer.parseInt(queryDTO.getMinMoodScore()));
        }
        if (StrUtil.isNotBlank(queryDTO.getMaxMoodScore())) {
            wrapper.le(EmotionDiary::getMoodScore, Integer.parseInt(queryDTO.getMaxMoodScore()));
        }
        if (StrUtil.isNotBlank(queryDTO.getDominantEmotion())) {
            wrapper.like(EmotionDiary::getDominantEmotion, queryDTO.getDominantEmotion());
        }

        wrapper.orderByDesc(EmotionDiary::getCreatedAt);

        Page<EmotionDiary> page = new Page<>(currentPage, size);
        Page<EmotionDiary> result = emotionDiaryMapper.selectPage(page, wrapper);

        List<EmotionDiaryResponseDTO> records = result.getRecords().stream()
                .map(diary -> {
                    User user = userMapper.selectById(diary.getUserId());
                    return EmotionDiaryResponseDTO.builder()
                            .id(diary.getId())
                            .userId(diary.getUserId())
                            .username(user != null ? user.getUsername() : null)
                            .nickname(user != null ? user.getNickname() : null)
                            .diaryDate(diary.getDiaryDate())
                            .moodScore(diary.getMoodScore())
                            .dominantEmotion(diary.getDominantEmotion())
                            .emotionTriggers(diary.getEmotionTriggers())
                            .diaryContent(diary.getDiaryContent())
                            .sleepQuality(diary.getSleepQuality())
                            .stressLevel(diary.getStressLevel())
                            .aiEmotionAnalysis(diary.getAiEmotionAnalysis())
                            .aiAnalysisUpdatedAt(diary.getAiAnalysisUpdatedAt())
                            .createdAt(diary.getCreatedAt())
                            .updatedAt(diary.getUpdatedAt())
                            .build();
                })
                .collect(Collectors.toList());

        Map<String, Object> data = new HashMap<>();
        data.put("records", records);
        data.put("total", result.getTotal());
        data.put("currentPage", currentPage);
        data.put("size", size);
        return data;
    }
}

package org.example.aispringboot.service;

import com.baomidou.mybatisplus.core.conditions.query.LambdaQueryWrapper;
import org.example.aispringboot.entity.KnowledgeCategory;
import org.example.aispringboot.mapper.KnowledgeCategoryMapper;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;

import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.stream.Collectors;

@Service
public class KnowledgeCategoryService {

    @Autowired
    private KnowledgeCategoryMapper knowledgeCategoryMapper;

    public List<Map<String, Object>> getCategoryTree() {
        // 查询所有启用的分类
        LambdaQueryWrapper<KnowledgeCategory> wrapper = new LambdaQueryWrapper<>();
        wrapper.eq(KnowledgeCategory::getStatus, 1)
                .orderByAsc(KnowledgeCategory::getSortOrder);
        List<KnowledgeCategory> allCategories = knowledgeCategoryMapper.selectList(wrapper);

        // 构建树形结构
        return buildTree(allCategories, 0L);
    }

    private List<Map<String, Object>> buildTree(List<KnowledgeCategory> allCategories, Long parentId) {
        return allCategories.stream()
                .filter(c -> parentId.equals(c.getParentId()))
                .map(category -> {
                    Map<String, Object> node = new HashMap<>();
                    node.put("id", category.getId());
                    node.put("categoryName", category.getCategoryName());
                    node.put("categoryCode", category.getCategoryCode());
                    node.put("description", category.getDescription());
                    node.put("sortOrder", category.getSortOrder());
                    node.put("children", buildTree(allCategories, category.getId()));
                    return node;
                })
                .collect(Collectors.toList());
    }
}

public KnowledgeArticleResponseDTO getArticleDetail(String id) {
    KnowledgeArticle article = knowledgeArticleMapper.selectById(id);
    if (article == null) {
        throw new RuntimeException("文章不存在");
    }

    // 增加阅读次数
    article.setReadCount(article.getReadCount() + 1);
    knowledgeArticleMapper.updateById(article);

    KnowledgeCategory category = knowledgeCategoryMapper.selectById(article.getCategoryId());
    User author = userMapper.selectById(article.getAuthorId());

    return KnowledgeArticleResponseDTO.builder()
            .id(article.getId())
            .categoryId(article.getCategoryId())
            .categoryName(category != null ? category.getCategoryName() : null)
            .title(article.getTitle())
            .summary(article.getSummary())
            .content(article.getContent())
            .coverImage(article.getCoverImage())
            .tags(article.getTags())
            .authorId(article.getAuthorId())
            .authorName(author != null ? author.getNickname() : null)
            .readCount(article.getReadCount())

package org.example.aispringboot.service;

import org.example.aispringboot.dto.response.FileUploadResponseDTO;
import org.example.aispringboot.entity.SysFileInfo;
import org.example.aispringboot.exception.BusinessException;
import org.example.aispringboot.mapper.SysFileInfoMapper;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Service;
import org.springframework.web.multipart.MultipartFile;

import java.io.File;
import java.io.IOException;
import java.time.LocalDateTime;
import java.util.UUID;

@Service
public class FileService {

    @Autowired
    private SysFileInfoMapper sysFileInfoMapper;

    @Value("${file.upload-path:./uploads/}")
    private String uploadPath;

    @Value("${file.base-url:http://localhost:1236/files/}")
    private String baseUrl;

    public FileUploadResponseDTO upload(MultipartFile file, String businessType, String businessId, String businessField, Long userId) {
        if (file.isEmpty()) {
            throw new BusinessException("文件不能为空");
        }

        // 获取文件信息
        String originalName = file.getOriginalFilename();
        String fileType = originalName.substring(originalName.lastIndexOf(".") + 1);
        long fileSize = file.getSize();

        // 生成唯一文件名
        String fileName = UUID.randomUUID().toString() + "." + fileType;

        // 创建上传目录（使用绝对路径）
        String datePath = LocalDateTime.now().toLocalDate().toString().replace("-", "/");
        File baseDir = new File(uploadPath).getAbsoluteFile();
        File dir = new File(baseDir, datePath);
        if (!dir.exists()) {
            dir.mkdirs();
        }

        // 保存文件
        File targetFile = new File(dir, fileName);
        try {
            file.transferTo(targetFile.toPath());
        } catch (IOException e) {
            throw new BusinessException("文件上传失败: " + e.getMessage());
        }
        String filePath = targetFile.getAbsolutePath();

        // 保存文件信息到数据库
        SysFileInfo fileInfo = new SysFileInfo();
        fileInfo.setOriginalName(originalName);
        fileInfo.setFilePath(filePath);
        fileInfo.setFileSize(fileSize);
        fileInfo.setFileType(fileType);
        fileInfo.setBusinessType(businessType);
        fileInfo.setBusinessId(businessId);
        fileInfo.setBusinessField(businessField);
        fileInfo.setUploadUserId(userId);
        fileInfo.setIsTemp(0);
        fileInfo.setStatus(1);
        fileInfo.setCreateTime(LocalDateTime.now());
        sysFileInfoMapper.insert(fileInfo);

        // 返回响应
        String url = baseUrl + datePath + "/" + fileName;
        return FileUploadResponseDTO.builder()
                .id(fileInfo.getId())
                .originalName(originalName)
                .filePath(filePath)
                .fileSize(fileSize)
                .fileType(fileType)
                .url(url)
                .build();
    }
}

public Map<String, Object> queryArticles(KnowledgeArticleQueryDTO queryDTO) {
    int currentPage = Integer.parseInt(queryDTO.getCurrentPage());
    int size = Integer.parseInt(queryDTO.getSize());

    LambdaQueryWrapper<KnowledgeArticle> wrapper = new LambdaQueryWrapper<>();

    if (StrUtil.isNotBlank(queryDTO.getTitle())) {
        wrapper.like(KnowledgeArticle::getTitle, queryDTO.getTitle());
    }
    if (StrUtil.isNotBlank(queryDTO.getCategoryId())) {
        wrapper.eq(KnowledgeArticle::getCategoryId, Long.parseLong(queryDTO.getCategoryId()));
    }
    if (StrUtil.isNotBlank(queryDTO.getStatus())) {
        wrapper.eq(KnowledgeArticle::getStatus, Integer.parseInt(queryDTO.getStatus()));
    }
    if (StrUtil.isNotBlank(queryDTO.getAuthorName())) {
        LambdaQueryWrapper<User> userWrapper = new LambdaQueryWrapper<>();
        userWrapper.like(User::getUsername, queryDTO.getAuthorName())
                .or()
                .like(User::getNickname, queryDTO.getAuthorName());
        List<User> users = userMapper.selectList(userWrapper);
        if (!users.isEmpty()) {
            List<Long> userIds = users.stream().map(User::getId).collect(Collectors.toList());
            wrapper.in(KnowledgeArticle::getAuthorId, userIds);
        } else {
            Map<String, Object> data = new HashMap<>();
            data.put("records", List.of());
            data.put("total", 0);
            data.put("currentPage", currentPage);
            data.put("size", size);
            return data;

public EmotionDiary createOrUpdateDiary(Long userId, EmotionDiaryCreateDTO dto) {
    // 检查该用户当天是否已有日记
    LambdaQueryWrapper<EmotionDiary> wrapper = new LambdaQueryWrapper<>();
    wrapper.eq(EmotionDiary::getUserId, userId)
            .eq(EmotionDiary::getDiaryDate, LocalDate.parse(dto.getDiaryDate()));
    EmotionDiary existingDiary = emotionDiaryMapper.selectOne(wrapper);

    if (existingDiary != null) {
        // 更新现有日记
        existingDiary.setMoodScore(dto.getMoodScore());
        existingDiary.setDominantEmotion(dto.getDominantEmotion());
        existingDiary.setEmotionTriggers(dto.getEmotionTriggers());
        existingDiary.setDiaryContent(dto.getDiaryContent());
        existingDiary.setSleepQuality(dto.getSleepQuality());
        existingDiary.setStressLevel(dto.getStressLevel());
        existingDiary.setUpdatedAt(LocalDateTime.now());
        emotionDiaryMapper.updateById(existingDiary);
        return existingDiary;
    } else {
        // 创建新日记
        EmotionDiary diary = new EmotionDiary();
        diary.setUserId(userId);
        diary.setDiaryDate(LocalDate.parse(dto.getDiaryDate()));
        diary.setMoodScore(dto.getMoodScore());
        diary.setDominantEmotion(dto.getDominantEmotion());
        diary.setEmotionTriggers(dto.getEmotionTriggers());
        diary.setDiaryContent(dto.getDiaryContent());
        diary.setSleepQuality(dto.getSleepQuality());
        diary.setStressLevel(dto.getStressLevel());
        diary.setCreatedAt(LocalDateTime.now());
        diary.setUpdatedAt(LocalDateTime.now());
        emotionDiaryMapper.insert(diary);
        return diary;
    }
}
